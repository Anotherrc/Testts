import React from 'react';
import type { Equipo } from '../types';

interface EquipoCardProps {
    equipo: Equipo;
    onActualizarHorometro?: (equipo: Equipo) => void;
    onDarDeBaja?: (equipo: Equipo) => void;
    onVerDetalle?: (equipo: Equipo) => void;
}

export const EquipoCard: React.FC<EquipoCardProps> = ({ equipo, onActualizarHorometro, onDarDeBaja, onVerDetalle }) => {
    const estadoVisual = equipo.estado === 'Mantenimiento' || equipo.color_semaforo === 'ROJO'
        ? { label: 'Urgente', className: 'bg-red-100 text-red-700 border-red-200' }
        : equipo.color_semaforo === 'AMARILLO'
            ? { label: 'Próximo', className: 'bg-amber-100 text-amber-700 border-amber-200' }
            : { label: 'Operativo', className: 'bg-emerald-100 text-emerald-700 border-emerald-200' };
    return (
        <div onClick={() => onVerDetalle?.(equipo)} className="bg-white rounded-2xl border border-slate-200 p-5 shadow-sm space-y-4 hover:-translate-y-0.5 hover:border-slate-300 hover:shadow-md transition-all cursor-pointer">
            <img
                src={equipo.foto_url || `https://images.unsplash.com/photo-${equipo.id % 2 ? '1581094794329-c8112a89af12' : '1504307651254-35680f356dfd'}?auto=format&fit=crop&w=800&q=80`}
                alt={equipo.nombre}
                className="h-32 w-full rounded-xl object-cover"
            />
            {/* Cabecera Tarjeta */}
            <div className="flex items-start justify-between">
                <div>
                    <span className="text-xs font-bold text-slate-400 uppercase tracking-wider">
                        {equipo.codigo}
                    </span>
                    <h3 className="text-base font-extrabold text-slate-900 mt-0.5">
                        {equipo.nombre}
                    </h3>
                </div>

                {/* Badge Semáforo */}
                <span className={`px-2.5 py-1 rounded-full text-xs font-extrabold uppercase border ${estadoVisual.className}`}>
                    {estadoVisual.label}
                </span>
            </div>

            {/* Datos de Horómetro y Metas */}
            <div className="grid grid-cols-2 gap-3 bg-slate-50 p-3 rounded-xl border border-slate-100 text-xs">
                <div>
                    <span className="text-slate-500 font-medium block">Horómetro Actual</span>
                    <span className="text-sm font-black text-slate-800">
                        {equipo.horometro_actual} {equipo.unidad_medida || 'hrs'}
                    </span>
                </div>
                <div>
                    <span className="text-slate-500 font-medium block">Uso Mes Actual</span>
                    <span className="text-sm font-black text-slate-800">
                        {equipo.horas_uso_mes} / {equipo.limite_contrato_mes} hrs
                    </span>
                </div>
            </div>

            {/* Botón Acción */}
            <div className="grid grid-cols-2 gap-2">
                {onActualizarHorometro && <button onClick={(event) => { event.stopPropagation(); onActualizarHorometro(equipo); }} className="rounded-xl bg-slate-900 px-3 py-2 text-xs font-bold text-white transition-colors hover:bg-slate-800">Actualizar lectura</button>}
                {onDarDeBaja && equipo.estado !== 'Deshabilitada' && <button onClick={(event) => { event.stopPropagation(); onDarDeBaja(equipo); }} className="rounded-xl border border-red-200 bg-red-50 px-3 py-2 text-xs font-bold text-red-700 transition-colors hover:bg-red-100">Dar de baja</button>}
            </div>
        </div>
    );
};
